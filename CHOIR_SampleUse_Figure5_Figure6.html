Figure 6: Time Series Plot Sample

data$year <- format(as.Date(data$date, format="%Y-%m-%d"),"%Y")
data$year <- as.numeric(data$year)
data <- data %>% filter(year>1984)

# To demonstrate a sample of increasing trends select COMIDs 10357898/3370264/19453135 in the Ohio River
OhioRiver1_annualmean <- data%>%
  filter(COMID==10357898)%>%
  group_by(year)%>%
  summarize(mean=mean(pred))

OhioRiver2_annualmean <- data %>%
  filter(COMID==3370264)%>%
  group_by(year)%>%
  summarize(mean=mean(pred))

OhioRiver3_annualmean <- data %>%
  filter(COMID==19453135)%>%
  group_by(year)%>%
  summarize(mean=mean(pred))

OhioRiver_annual <- ggplot()+
  geom_line(OhioRiver1_annualmean, mapping=aes(x=year, y=mean), col='lightgrey', alpha=0.8, linewidth=1)+
  geom_line(OhioRiver2_annualmean, mapping=aes(x=year, y=mean), col='#515151', alpha=0.8, linewidth=1)+
  geom_line(OhioRiver3_annualmean, mapping=aes(x=year, y=mean), col='black', alpha=0.8, linewidth=1)+
  xlab("Year")+
  ylab("Chl-a Conc. (ug/L)")+
  labs(title="Ohio River")+
  theme_classic()+
  theme(axis.text=element_text(size=14))
OhioRiver_annual
